\begin{abstract}
    Assignment used to provide further practice on factoring polynomials with greatest common factor (GCF) and by grouping.
\end{abstract}

\begin{problem}
    Factoring a polynomial with a GCF: $$-36x^3+42x^2-6x$$
    $\answer{6x(-6x^2+7x-1)}$
\end{problem}

\begin{problem}
    Factoring a polynomial with a GCF: $$16x^4y-14x^5y^2+22x^4y^4-2x^6y^2$$
    
    $\answer{2x^3y(8x-7x^2y+11xy^3-x^3y)}$
\end{problem}

\begin{problem}
    Factoring a polynomial with a GCF: $$90z^3-105x^2-72z+84$$
    
    $\answer{3(6z-7)(5z^2-4)}$
\end{problem}

\begin{problem}
    Factoring a polynomial with a GCF: $$4ay+20a^2-8by-40ab$$
    
    $\answer{4(y+5a)(a-2b)}$
\end{problem}
