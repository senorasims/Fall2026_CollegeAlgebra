\begin{abstract}
    Assignment used to provide further practice on factoring trinomials.
\end{abstract}

\begin{problem}
Factor the following the trinomial $x^2-3x-54$.

$\answer{(x-9)(x+6)}$
\end{problem}

\begin{problem}
Factor the following the trinomial
$x^2+9x-36$.

$\answer{(x+12)(x-3)}$
\end{problem}

\begin{problem}
Factor the following the trinomial $3x^2+16x+5$.

$\answer{(3x+1)(x+5)}$
\end{problem}

\begin{problem}
Factor the following the trinomial $-20x^3+74x-60x$.

$\answer{-2x(5x-6)(2x-5)}$
\end{problem}
