\begin{abstract}
    Assignment used to provide further practice on factoring polynomials with special factoring patterns.
\end{abstract}

\section{Difference of Squares}

\begin{problem}
    Factor $m^2-64$.

    $\answer{(m+8)(m-8)}$
\end{problem}

\begin{problem}
    Factor $m^4-16$.

    $\answer{(m^2+4)(m-2)(m+2)}$
\end{problem}

\section{Perfect Square Trinomials}

\begin{problem}
    Factor $x^2+26x+169$.

    $\answer{(x+13)(x+13)}$
\end{problem}

\begin{problem}
    Factor $2x^2-8x+8$.

    $\answer{2(x-2)(x-2) }$
\end{problem}
